cmake_minimum_required(VERSION 3.21)
project(SavageModPlayerDesktop LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../VERSION")
file(STRINGS "../../VERSION" SAVAGE_VERSION LIMIT_COUNT 1)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 DBus Network Multimedia)
qt_add_executable(savage-mod-player main.cpp VideoRenderer.cpp MediaControls.cpp UpdateController.cpp)
target_compile_definitions(savage-mod-player PRIVATE SAVAGE_VERSION="${SAVAGE_VERSION}")
target_link_libraries(savage-mod-player PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::Network)
qt_add_resources(savage-mod-player interface PREFIX "/" FILES
    Main.qml PlayerButton.qml PlayerIcon.qml PlayerCheckBox.qml PlayerSlider.qml ChannelTag.qml MarqueeLabel.qml ScopeView.qml TrackerView.qml NoteRain.qml SampleWaveform.qml InstrumentDetails.qml FormButton.qml FormCombo.qml CompactDashboard.qml)
install(TARGETS savage-mod-player RUNTIME DESTINATION bin)
qt_add_resources(savage-mod-player applicationIcon PREFIX "/" BASE "../../src" FILES "../../src/AppIcon.png")

foreach(helper update.py install.py)
    configure_file("../${helper}" "${CMAKE_CURRENT_BINARY_DIR}/${helper}" COPYONLY)
endforeach()
configure_file("../../VERSION" "${CMAKE_CURRENT_BINARY_DIR}/VERSION" COPYONLY)
configure_file("../../config/sparkle-public-key.txt" "${CMAKE_CURRENT_BINARY_DIR}/update-public-key.txt" COPYONLY)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(savage-update-tests ../../Tests/qt_update_tests.cpp UpdateController.cpp)
    target_include_directories(savage-update-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(savage-update-tests PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME updater-process COMMAND savage-update-tests)
endif()

configure_file("../licenses/Swift-LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/Swift-LICENSE.txt" COPYONLY)
